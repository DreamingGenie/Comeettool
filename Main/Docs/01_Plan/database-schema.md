# 데이터베이스 스키마 (ERD DDL)

ERD를 기반으로 작성한 DDL이다. 테이블·제약조건·외래키 정의를 담는다.
ERD 시각 다이어그램은 별도 도구에서 관리하며, 이 문서는 그 산출물(DDL)이다.

> **참고.** 아래 DDL은 **PostgreSQL** 문법이다(`UUID`, `BYTEA`, `JSONB`, `TIMESTAMPTZ`, `BIGINT[]`).
> 핵심 도메인 DB로 PostgreSQL 사용 확정 ([architecture.md](architecture.md) §3).
> PK는 `BIGINT GENERATED BY DEFAULT AS IDENTITY`로 자동 채번한다(엔티티 `@GeneratedValue(strategy = IDENTITY)`와 정합). UUID PK는 애플리케이션에서 생성한다.

> **역할/권한 모델 (S15P11A707-28 반영).**
> - 권한은 `members.authority` 고정 3종: `OWNER` / `MEMBER` / `GUEST`.
> - 역할은 팀이 정의하는 커스텀 값으로 `team_roles`(카탈로그, 컬럼 `role_name`)에 두고, 멤버는 `members.team_role_id`(FK, NULL 허용, `ON DELETE SET NULL`)로 **1개**를 참조한다(1 멤버 = 최대 1 역할).
> - 회의 참여자 테이블은 `participants`(구 `meeting_member`)다.

## DDL

```sql
-- 기존 테이블 삭제 (있으면 삭제, 없으면 무시)
DROP TABLE IF EXISTS documents_state CASCADE;
DROP TABLE IF EXISTS chats CASCADE;
DROP TABLE IF EXISTS members CASCADE;
DROP TABLE IF EXISTS team_roles CASCADE;
DROP TABLE IF EXISTS users CASCADE;
DROP TABLE IF EXISTS teams CASCADE;
DROP TABLE IF EXISTS schedules CASCADE;
DROP TABLE IF EXISTS alarms CASCADE;
DROP TABLE IF EXISTS documents CASCADE;
DROP TABLE IF EXISTS sound_texts CASCADE;
DROP TABLE IF EXISTS minutes CASCADE;
DROP TABLE IF EXISTS meeting_rooms CASCADE;
DROP TABLE IF EXISTS participants CASCADE;
DROP TABLE IF EXISTS meeting_member CASCADE;  -- 구 명칭(→ participants). 기존 스키마 재적용 시 정리용
DROP TABLE IF EXISTS documents_version CASCADE;
DROP TABLE IF EXISTS voices CASCADE;

CREATE TABLE documents_state (
    document_id UUID NOT NULL,
    yjs_state BYTEA NOT NULL,
    binary_size INTEGER NOT NULL,
    state_hash BYTEA NOT NULL,
    schema_version INTEGER NOT NULL DEFAULT 1,
    persisted_revision BIGINT NOT NULL DEFAULT 0,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE chats (
    chat_id BIGINT GENERATED BY DEFAULT AS IDENTITY,
    meeting_room_id BIGINT NOT NULL,
    team_id BIGINT NOT NULL,
    date TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    chat TEXT NOT NULL
);

-- role 컬럼 제거. authority = 고정 권한(OWNER/MEMBER/GUEST).
-- team_role_id = 배정 역할(1 멤버 = 최대 1 역할, 미배정 시 NULL).
CREATE TABLE members (
    member_id BIGINT GENERATED BY DEFAULT AS IDENTITY,
    user_id BIGINT NOT NULL,
    team_id BIGINT NOT NULL,
    authority VARCHAR NOT NULL,
    team_role_id BIGINT NULL,
    nickname VARCHAR NOT NULL
);

-- 팀이 정의하는 역할 카탈로그 (owner가 생성). 역할 이름의 단일 원천.
CREATE TABLE team_roles (
    team_role_id BIGINT GENERATED BY DEFAULT AS IDENTITY,
    team_id BIGINT NOT NULL,
    role_name VARCHAR NOT NULL,
    color VARCHAR NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE users (
    user_id BIGINT GENERATED BY DEFAULT AS IDENTITY,
    password_hash VARCHAR(255) NOT NULL,
    nickname VARCHAR NULL,
    user_profile_image VARCHAR NULL,
    email VARCHAR NOT NULL,
    phone VARCHAR NULL,
    job_family VARCHAR NULL,
    job_role VARCHAR NULL,
    user_description TEXT NULL,
    sex VARCHAR NULL,
    age INTEGER NULL,
    user_color VARCHAR NOT NULL DEFAULT '#000000',
    space_order TEXT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMPTZ NULL,
    is_deleted BOOLEAN NOT NULL DEFAULT FALSE
);

CREATE TABLE teams (
    team_id BIGINT GENERATED BY DEFAULT AS IDENTITY,
    team_name VARCHAR NOT NULL,
    team_description TEXT NULL,
    team_owner_id BIGINT NOT NULL,
    team_profile_image VARCHAR NULL,
    team_color VARCHAR NOT NULL DEFAULT '#000000',
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMPTZ NULL,
    is_deleted BOOLEAN NOT NULL DEFAULT FALSE,
    team_invite_link VARCHAR NULL
);

CREATE TABLE schedules (
    schedule_id BIGINT GENERATED BY DEFAULT AS IDENTITY,
    team_id BIGINT NOT NULL,
    schedule_category VARCHAR(1000) NULL,
    schedule_title VARCHAR(1000) NULL,
    schedule_description TEXT NULL,
    start_time TIMESTAMPTZ NULL,
    end_time TIMESTAMPTZ NULL,
    user_id_arr BIGINT[] NULL
);

CREATE TABLE alarms (
    alarm_id BIGINT GENERATED BY DEFAULT AS IDENTITY,
    message TEXT NOT NULL,
    alarm_category VARCHAR NOT NULL,
    alarm_property VARCHAR NOT NULL,
    user_id BIGINT NOT NULL
);

CREATE TABLE documents (
    document_id UUID NOT NULL,
    team_id BIGINT NOT NULL,
    document_title VARCHAR(1000) NOT NULL DEFAULT '새 문서',
    final_version INTEGER NOT NULL DEFAULT 0,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMPTZ NULL,
    is_deleted BOOLEAN NOT NULL DEFAULT FALSE
);

CREATE TABLE sound_texts (
    sound_text_id BIGINT GENERATED BY DEFAULT AS IDENTITY,
    meeting_room_id BIGINT NOT NULL,
    team_id BIGINT NOT NULL,
    date TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    sound_text TEXT NOT NULL
);

CREATE TABLE minutes (
    minute_id BIGINT GENERATED BY DEFAULT AS IDENTITY,
    meeting_room_id BIGINT NOT NULL,
    team_id BIGINT NOT NULL,
    date TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    minute TEXT NOT NULL,
    is_checked BOOLEAN NOT NULL
);

CREATE TABLE meeting_rooms (
    meeting_room_id BIGINT GENERATED BY DEFAULT AS IDENTITY,
    team_id BIGINT NOT NULL,
    host_id BIGINT NOT NULL,
    meeting_room_name VARCHAR(250) NOT NULL DEFAULT '새 회의',
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMPTZ NULL,
    is_deleted BOOLEAN NOT NULL DEFAULT FALSE
);

CREATE TABLE participants (
    participant_id BIGINT GENERATED BY DEFAULT AS IDENTITY,
    meeting_room_id BIGINT NOT NULL,
    member_id BIGINT NOT NULL,
    participants_role VARCHAR(50) NOT NULL,
    is_host BOOLEAN NOT NULL
);

CREATE TABLE documents_version (
    document_version_id UUID NOT NULL,
    document_id UUID NOT NULL,
    version_number INTEGER NOT NULL,
    title_snapshot VARCHAR(1000) NOT NULL DEFAULT '새 문서',
    yjs_state BYTEA NOT NULL,
    editor_json JSONB NOT NULL,
    trigger_type VARCHAR(30) NOT NULL,
    binary_size INTEGER NOT NULL,
    state_hash BYTEA NOT NULL,
    schema_version INTEGER NOT NULL DEFAULT 1,
    source_version_id UUID NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    request_id UUID NULL
);

CREATE TABLE voices (
    voice_id BIGINT GENERATED BY DEFAULT AS IDENTITY,
    meeting_room_id BIGINT NOT NULL,
    team_id BIGINT NOT NULL,
    date TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    voice BYTEA NOT NULL
);

-- ===== Primary Keys =====
ALTER TABLE documents_state ADD CONSTRAINT PK_DOCUMENTS_STATE PRIMARY KEY (document_id);
ALTER TABLE chats ADD CONSTRAINT PK_CHATS PRIMARY KEY (chat_id);
ALTER TABLE members ADD CONSTRAINT PK_MEMBERS PRIMARY KEY (member_id);
ALTER TABLE team_roles ADD CONSTRAINT PK_TEAM_ROLES PRIMARY KEY (team_role_id);
ALTER TABLE users ADD CONSTRAINT PK_USERS PRIMARY KEY (user_id);
ALTER TABLE teams ADD CONSTRAINT PK_TEAMS PRIMARY KEY (team_id);
ALTER TABLE schedules ADD CONSTRAINT PK_SCHEDULES PRIMARY KEY (schedule_id);
ALTER TABLE alarms ADD CONSTRAINT PK_ALARMS PRIMARY KEY (alarm_id);
ALTER TABLE documents ADD CONSTRAINT PK_DOCUMENTS PRIMARY KEY (document_id);
ALTER TABLE sound_texts ADD CONSTRAINT PK_SOUND_TEXTS PRIMARY KEY (sound_text_id);
ALTER TABLE minutes ADD CONSTRAINT PK_MINUTES PRIMARY KEY (minute_id);
ALTER TABLE meeting_rooms ADD CONSTRAINT PK_MEETING_ROOMS PRIMARY KEY (meeting_room_id);
ALTER TABLE participants ADD CONSTRAINT PK_PARTICIPANTS PRIMARY KEY (participant_id);
ALTER TABLE documents_version ADD CONSTRAINT PK_DOCUMENTS_VERSION PRIMARY KEY (document_version_id);
ALTER TABLE voices ADD CONSTRAINT PK_VOICES PRIMARY KEY (voice_id);

-- ===== Unique Keys / Check Constraints =====
ALTER TABLE participants
    ADD CONSTRAINT UK_PARTICIPANTS_ROOM_MEMBER
    UNIQUE (meeting_room_id, member_id);

ALTER TABLE members
    ADD CONSTRAINT UK_MEMBERS_TEAM_USER
    UNIQUE (team_id, user_id);

ALTER TABLE team_roles
    ADD CONSTRAINT UK_TEAM_ROLES_TEAM_NAME
    UNIQUE (team_id, role_name);

ALTER TABLE users
    ADD CONSTRAINT UK_USERS_EMAIL
    UNIQUE (email);

ALTER TABLE users
    ADD CONSTRAINT CK_USERS_NICKNAME_LENGTH
    CHECK (char_length(nickname) <= 20);

ALTER TABLE users
    ADD CONSTRAINT CK_USERS_SEX
    CHECK (sex IN ('M', 'F'));

ALTER TABLE users
    ADD CONSTRAINT CK_USERS_AGE
    CHECK (age IN (10, 20, 30, 40, 50, 60));

ALTER TABLE documents_version
    ADD CONSTRAINT UK_DOCUMENTS_VERSION_NUMBER
    UNIQUE (document_id, version_number);

ALTER TABLE documents_version
    ADD CONSTRAINT UK_DOCUMENTS_VERSION_REQUEST
    UNIQUE (document_id, request_id);

ALTER TABLE documents_state
    ADD CONSTRAINT CK_DOCUMENTS_STATE_HASH_LENGTH
    CHECK (octet_length(state_hash) = 32);

ALTER TABLE documents_state
    ADD CONSTRAINT CK_DOCUMENTS_STATE_BINARY_SIZE
    CHECK (binary_size = octet_length(yjs_state));

ALTER TABLE documents_state
    ADD CONSTRAINT CK_DOCUMENTS_STATE_SCHEMA_VERSION
    CHECK (schema_version > 0);

ALTER TABLE documents_state
    ADD CONSTRAINT CK_DOCUMENTS_STATE_PERSISTED_REVISION
    CHECK (persisted_revision >= 0);

ALTER TABLE documents_version
    ADD CONSTRAINT CK_DOCUMENTS_VERSION_HASH_LENGTH
    CHECK (octet_length(state_hash) = 32);

ALTER TABLE documents_version
    ADD CONSTRAINT CK_DOCUMENTS_VERSION_NUMBER
    CHECK (version_number > 0);

ALTER TABLE documents_version
    ADD CONSTRAINT CK_DOCUMENTS_VERSION_BINARY_SIZE
    CHECK (binary_size = octet_length(yjs_state));

ALTER TABLE documents_version
    ADD CONSTRAINT CK_DOCUMENTS_VERSION_SCHEMA_VERSION
    CHECK (schema_version > 0);

ALTER TABLE documents
    ADD CONSTRAINT CK_DOCUMENTS_FINAL_VERSION
    CHECK (final_version >= 0);

-- ===== Foreign Keys =====
ALTER TABLE documents_state ADD CONSTRAINT FK_DOCUMENTS_TO_DOCUMENTS_STATE_1
    FOREIGN KEY (document_id) REFERENCES documents (document_id)
    ON DELETE CASCADE;

ALTER TABLE chats ADD CONSTRAINT FK_MEETING_ROOMS_TO_CHATS_1
    FOREIGN KEY (meeting_room_id) REFERENCES meeting_rooms (meeting_room_id);

ALTER TABLE members ADD CONSTRAINT FK_USERS_TO_MEMBERS_1
    FOREIGN KEY (user_id) REFERENCES users (user_id);

ALTER TABLE members ADD CONSTRAINT FK_TEAMS_TO_MEMBERS_1
    FOREIGN KEY (team_id) REFERENCES teams (team_id);

-- members → team_roles (역할 삭제 시 멤버는 유지, 배정만 해제 → SET NULL)
ALTER TABLE members ADD CONSTRAINT FK_TEAM_ROLES_TO_MEMBERS_1
    FOREIGN KEY (team_role_id) REFERENCES team_roles (team_role_id)
    ON DELETE SET NULL;

ALTER TABLE team_roles ADD CONSTRAINT FK_TEAMS_TO_TEAM_ROLES_1
    FOREIGN KEY (team_id) REFERENCES teams (team_id);

ALTER TABLE schedules ADD CONSTRAINT FK_TEAMS_TO_SCHEDULES_1
    FOREIGN KEY (team_id) REFERENCES teams (team_id);

ALTER TABLE alarms ADD CONSTRAINT FK_USERS_TO_ALARMS_1
    FOREIGN KEY (user_id) REFERENCES users (user_id);

ALTER TABLE documents ADD CONSTRAINT FK_TEAMS_TO_DOCUMENTS_1
    FOREIGN KEY (team_id) REFERENCES teams (team_id);

ALTER TABLE sound_texts ADD CONSTRAINT FK_MEETING_ROOMS_TO_SOUND_TEXTS_1
    FOREIGN KEY (meeting_room_id) REFERENCES meeting_rooms (meeting_room_id);

ALTER TABLE minutes ADD CONSTRAINT FK_MEETING_ROOMS_TO_MINUTES_1
    FOREIGN KEY (meeting_room_id) REFERENCES meeting_rooms (meeting_room_id);

ALTER TABLE meeting_rooms ADD CONSTRAINT FK_TEAMS_TO_MEETING_ROOMS_1
    FOREIGN KEY (team_id) REFERENCES teams (team_id);

ALTER TABLE participants ADD CONSTRAINT FK_MEETING_ROOMS_TO_PARTICIPANTS_1
    FOREIGN KEY (meeting_room_id) REFERENCES meeting_rooms (meeting_room_id);

ALTER TABLE participants ADD CONSTRAINT FK_MEMBERS_TO_PARTICIPANTS_1
    FOREIGN KEY (member_id) REFERENCES members (member_id);

ALTER TABLE documents_version ADD CONSTRAINT FK_DOCUMENTS_TO_DOCUMENTS_VERSION_1
    FOREIGN KEY (document_id) REFERENCES documents (document_id)
    ON DELETE CASCADE;

ALTER TABLE documents_version ADD CONSTRAINT FK_DOCUMENTS_VERSION_SOURCE
    FOREIGN KEY (source_version_id)
    REFERENCES documents_version (document_version_id)
    ON DELETE SET NULL;

ALTER TABLE voices ADD CONSTRAINT FK_MEETING_ROOMS_TO_VOICES_1
    FOREIGN KEY (meeting_room_id) REFERENCES meeting_rooms (meeting_room_id);

-- ===== Indexes =====
CREATE INDEX IDX_CHATS_MEETING_ROOM_ID
    ON chats (meeting_room_id);

CREATE INDEX IDX_MEMBERS_USER_ID
    ON members (user_id);

CREATE INDEX IDX_TEAM_ROLES_TEAM_ID
    ON team_roles (team_id);

CREATE INDEX IDX_MEMBERS_TEAM_ROLE_ID
    ON members (team_role_id);

CREATE INDEX IDX_SCHEDULES_TEAM_ID
    ON schedules (team_id);

CREATE INDEX IDX_ALARMS_USER_ID
    ON alarms (user_id);

CREATE INDEX IDX_DOCUMENTS_TEAM_ID
    ON documents (team_id);

CREATE INDEX IDX_SOUND_TEXTS_MEETING_ROOM_ID
    ON sound_texts (meeting_room_id);

CREATE INDEX IDX_MINUTES_MEETING_ROOM_ID
    ON minutes (meeting_room_id);

CREATE INDEX IDX_MEETING_ROOMS_TEAM_ID
    ON meeting_rooms (team_id);

CREATE INDEX IDX_PARTICIPANTS_MEMBER_ID
    ON participants (member_id);

CREATE INDEX IDX_DOCUMENTS_VERSION_SOURCE_VERSION_ID
    ON documents_version (source_version_id);

CREATE INDEX IDX_VOICES_MEETING_ROOM_ID
    ON voices (meeting_room_id);
```
